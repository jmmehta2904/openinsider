OpenInsider Important Commands
==============================

This file is a working command reference for local development, GCP setup,
Cloud Functions, Airflow, and cost control. Do not paste real secrets into this
file.

---------

Deploy Cloud Functions
----------------------

gcloud functions deploy fetch_macro \
  --gen2 \
  --runtime python312 \
  --region "$REGION" \
  --entry-point fetch_macro \
  --trigger-http \
  --no-allow-unauthenticated \
  --service-account "openinsider-ingestion-sa@$PROJECT_ID.iam.gserviceaccount.com" \
  --memory 256MB \
  --timeout 180s \
  --set-env-vars "GCS_BUCKET=$GCS_BUCKET,GCP_PROJECT_ID=$PROJECT_ID" \
  --set-secrets "FRED_KEY=FRED_KEY:latest" \
  --source /tmp/openinsider-functions/fetch_macro

gcloud functions deploy fetch_prices \
  --gen2 \
  --runtime python312 \
  --region "$REGION" \
  --entry-point fetch_prices \
  --trigger-http \
  --no-allow-unauthenticated \
  --service-account "openinsider-ingestion-sa@$PROJECT_ID.iam.gserviceaccount.com" \
  --memory 512MB \
  --timeout 180s \
  --set-env-vars "GCS_BUCKET=$GCS_BUCKET,GCP_PROJECT_ID=$PROJECT_ID" \
  --set-secrets "FINNHUB_KEY=FINNHUB_KEY:latest" \
  --source /tmp/openinsider-functions/fetch_prices

gcloud functions deploy fetch_filings \
  --gen2 \
  --runtime python312 \
  --region "$REGION" \
  --entry-point fetch_filings \
  --trigger-http \
  --no-allow-unauthenticated \
  --service-account "openinsider-ingestion-sa@$PROJECT_ID.iam.gserviceaccount.com" \

Airflow VM
----------

  --project="$PROJECT_ID" \
  --zone="$ZONE" \
  --machine-type=e2-micro \
  --image-family=ubuntu-2404-lts-amd64 \
  --image-project=ubuntu-os-cloud \
  --boot-disk-size=30GB \
  --service-account="$AIRFLOW_SA" \
  --scopes=https://www.googleapis.com/auth/cloud-platform \
  --tags=airflow-vm

gcloud compute ssh "$VM_NAME" --zone="$ZONE"

Cost Control: GCS Lifecycle
---------------------------

cat > lifecycle.json <<'EOF'
{
  "rule": [
    {
      "action": {"type": "Delete"},
      "condition": {
        "age": 14,
        "matchesPrefix": ["filings/", "prices/", "news/", "macro/", "reports/"]
      }
    }
  ]
}
EOF

gcloud storage buckets update "gs://$GCS_BUCKET" --lifecycle-file=lifecycle.json
